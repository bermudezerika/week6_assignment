def get_fred(series_id):
    """Download a FRED series (fredapi if FRED_API_KEY is set, otherwise the public CSV endpoint)."""
    key = os.environ.get("FRED_API_KEY")
    if key:
        from fredapi import Fred
        s = Fred(api_key=key).get_series(series_id)
    else:
        url = f"https://fred.stlouisfed.org/graph/fredgraph.csv?id={series_id}"
        raw = pd.read_csv(url, parse_dates=["observation_date"], index_col="observation_date")
        s = pd.to_numeric(raw[series_id], errors="coerce")
    return s.dropna().rename(series_id)

LAG = 12                                   # months; reflects annual lease turnover
COVID = ("2020-03-01", "2021-12-01")
HAC = dict(cov_type="HAC", cov_kwds={"maxlags": 12})

real = get_fred("FII10")
rent = get_fred("CUURA422SEHA")
rent = rent[rent.index >= "2002-01-01"]
missing_months = pd.date_range(rent.index.min(), rent.index.max(), freq="MS").difference(rent.index)

rent_yoy = (rent / rent.reindex(rent.index - pd.DateOffset(months=12)).values - 1) * 100
df = pd.DataFrame({"rent_yoy": rent_yoy}).join(real.rename("real"), how="inner")
df["real_chg"] = df["real"] - real.reindex(df.index - pd.DateOffset(months=12)).values
df["real_lag"] = real.reindex(df.index - pd.DateOffset(months=LAG)).values
df["covid"] = ((df.index >= COVID[0]) & (df.index <= COVID[1])).astype(int)
df = df.dropna()
N, start, end = len(df), df.index.min(), df.index.max()

specs = [("(1) Level", "rent_yoy ~ real", "real"),
         ("(2) Level + COVID", "rent_yoy ~ real + covid", "real"),
         ("(3) 12-month change", "rent_yoy ~ real_chg", "real_chg"),
         ("(4) 12-month change + COVID", "rent_yoy ~ real_chg + covid", "real_chg"),
         (f"(5) Level, {LAG}-month lag + COVID", "rent_yoy ~ real_lag + covid", "real_lag")]
models = {name: (smf.ols(f, df).fit(**HAC), var) for name, f, var in specs}
def est(name):
    m, var = models[name]
    return m.params[var], m.bse[var], m.pvalues[var]

med = df["real"].median()
hi, lo = df.loc[df["real"] > med, "rent_yoy"], df.loc[df["real"] <= med, "rent_yoy"]
ttest = stats.ttest_ind(hi, lo, equal_var=False)

lags = range(0, 25, 2)
lag_corr = pd.Series([df["rent_yoy"].corr(real.reindex(df.index - pd.DateOffset(months=k)).set_axis(df.index)) for k in lags],
                     index=lags)

def p_txt(p): return "p < 0.001" if p < 0.001 else f"p = {p:.3f}"
def verdict(b, p):
    if p >= 0.05: return "Not significant"
    return "Supports H1 (+)" if b > 0 else "Supports H2 (−)"

b1, se1, p1 = est("(1) Level")
b2, se2, p2 = est("(2) Level + COVID")
b4, se4, p4 = est("(4) 12-month change + COVID")
b5, se5, p5 = est(f"(5) Level, {LAG}-month lag + COVID")
verdicts = [verdict(b2, p2), verdict(b4, p4), verdict(b5, p5), verdict(hi.mean() - lo.mean(), ttest.pvalue)]
n_h1, n_h2 = sum(v.startswith("Supports H1") for v in verdicts), sum(v.startswith("Supports H2") for v in verdicts)
finding = ("H2" if n_h2 > n_h1 else "H1" if n_h1 > n_h2 else None)

1. Introduction¶

This report examines whether demand for San Francisco rental housing, measured by rent growth, rises when real (inflation-adjusted) long-term interest rates are higher. Using a real rather than nominal rate separates the cost of borrowing from general inflation, which otherwise moves both interest rates and rents.

Two competing hypotheses guide the analysis:

  • H1 (positive): higher real rates raise the cost of home ownership. More households continue renting, which increases rental demand and rent growth.
  • H2 (negative): higher real rates tighten financial conditions and slow employment growth, particularly in San Francisco's technology sector, which reduces rental demand and rent growth.

3. Methodology¶

Variables

  • Rent growth: year-over-year percentage change in the rent index, comparing each month with the same month a year earlier.
  • Real rate: the 10-year TIPS yield, measured three ways: its level, its change over the prior 12 months, and its level 12 months earlier. The 12-month lag reflects the annual turnover of residential leases, through which changes in housing demand reach measured rents.
  • COVID indicator: equals 1 from March 2020 through December 2021, when pandemic-related migration out of San Francisco depressed rents independently of interest rates.

Tests

  1. Ordinary least squares regressions of rent growth on each real-rate measure, with and without the COVID indicator. Year-over-year growth rates overlap and are persistent, so inference uses Newey-West (HAC) standard errors with 12 lags.
  2. A comparison of mean rent growth in months with above- versus below-median real rates (Welch two-sample t-test).
  3. A profile of correlations between rent growth and the real rate at lags of 0 to 24 months.

All tests use the same sample. Results are evaluated at the 5% significance level.

6. Limitations¶

  • Short sample. About two decades of data cover only a few independent cycles (the 2008 financial crisis, the 2010s technology boom, the pandemic, and the 2022 rate increases), so a small number of episodes drive the results.
  • Correlation, not causation. Monetary policy responds to the same business cycle that drives local labor markets, so both series may reflect a common cause.
  • Measurement. The rent index includes continuing leases, so it lags changes in market rents. It also measures the price of rental housing rather than the quantity demanded.
